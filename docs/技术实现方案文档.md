#### 3.1.1 API端点映射

| 功能 | GitHub API端点 | 说明 |

#### 3.1.3 速率限制处理

- **认证后限制**：5000次/小时
- **未认证限制**：60次/小时
- **实现策略**：
  - 使用Redis缓存API响应
  - 实现指数退避重试
  - 监控`X-RateLimit-Remaining`响应头

```python
class GitHubService:
    def __init__(self, token: str):
        self.token = token
        self.base_url = "https://api.github.com"
        self.session = aiohttp.ClientSession()
        self.cache = RedisCache(ttl=3600)  # 1小时缓存

    async def get_repo_info(self, owner: str, repo: str) -> dict:
        cache_key = f"repo:{owner}/{repo}"
        cached = await self.cache.get(cache_key)
        if cached:
            return cached

        url = f"{self.base_url}/repos/{owner}/{repo}"
        async with self.session.get(url, headers=self._headers()) as resp:
            if resp.status == 403 and "rate limit" in (await resp.text()).lower():
                raise RateLimitError("GitHub API rate limit exceeded")
            data = await resp.json()
            await self.cache.set(cache_key, data)
            return data
```

### 3.2 AI模型多模型支持

#### 3.2.1 模型抽象层设计

```python
from abc import ABC, abstractmethod
from typing import AsyncGenerator

class AIModelProvider(ABC):
    """AI模型提供者抽象基类"""

    @abstractmethod
    async def generate(self, prompt: str, **kwargs) -> str:
        """生成文本"""
        pass

    @abstractmethod
    async def generate_stream(self, prompt: str, **kwargs) -> AsyncGenerator[str, None]:
        """流式生成文本"""
        pass

class ClaudeProvider(AIModelProvider):
    """Claude模型提供者"""

    def __init__(self, api_key: str):
        import anthropic
        self.client = anthropic.AsyncAnthropic(api_key=api_key)

    async def generate(self, prompt: str, **kwargs) -> str:
        response = await self.client.messages.create(
            model="claude-sonnet-4-20250514",
            max_tokens=kwargs.get("max_tokens", 4096),
            messages=[{"role": "user", "content": prompt}]
        )
        return response.content[0].text

    async def generate_stream(self, prompt: str, **kwargs) -> AsyncGenerator[str, None]:
        async with self.client.messages.stream(
            model="claude-sonnet-4-20250514",
            max_tokens=kwargs.get("max_tokens", 4096),
            messages=[{"role": "user", "content": prompt}]
        ) as stream:
            async for text in stream.text_stream:
                yield text

class OpenAIProvider(AIModelProvider):
    """OpenAI模型提供者"""
    # 类似实现...

class QwenProvider(AIModelProvider):
    """通义千问模型提供者"""
    # 类似实现...

class AIServiceFactory:
```

{content}

#### 3.3.2 翻译流程

```python
class TranslateService:
    def __init__(self, ai_provider: AIModelProvider):
        self.ai = ai_provider

    async def translate_readme(self, content: str) -> str:
        # 1. 预处理：提取代码块
        code_blocks, processed_content = self._extract_code_blocks(content)

        # 2. 翻译（保留代码块占位符）
        prompt = TRANSLATE_PROMPT.format(content=processed_content)
        translated = await self.ai.generate(prompt)

        # 3. 后处理：还原代码块
        result = self._restore_code_blocks(translated, code_blocks)

        return result

    def _extract_code_blocks(self, content: str) -> tuple[list[str], str]:
        """提取代码块，用占位符替换"""
        import re
        code_blocks = []
        pattern = r'```[\s\S]*?```'

        def replace(match):
            code_blocks.append(match.group(0))
            return f"__CODE_BLOCK_{len(code_blocks)-1}__"

        processed = re.sub(pattern, replace, content)
        return code_blocks, processed

    def _restore_code_blocks(self, content: str, code_blocks: list[str]) -> str:
        """还原代码块"""
        for i, block in enumerate(code_blocks):
            content = content.replace(f"__CODE_BLOCK_{i}__", block)
        return content
```

```python
class TechStackService:
    def __init__(self, github_service: GitHubService):
        self.github = github_service

    async def analyze(self, owner: str, repo: str) -> TechStackResult:
        # 1. 获取语言分布
        languages = await self.github.get_languages(owner, repo)

        # 2. 获取Topics标签
        topics = await self.github.get_topics(owner, repo)

        # 3. 获取依赖文件
        dependencies = await self._get_dependencies(owner, repo)

        # 4. 分析框架和工具
        frameworks = self._detect_frameworks(dependencies, topics)
        tools = self._detect_tools(dependencies)

        return TechStackResult(
            languages=self._calculate_percentages(languages),
            frameworks=frameworks,
            tools=tools,
            databases=self._detect_databases(dependencies),
            cloud_services=self._detect_cloud_services(dependencies)
        )

    async def _get_dependencies(self, owner: str, repo: str) -> dict:
        """获取依赖文件内容"""
        result = {}
        dep_files = [
            "package.json", "requirements.txt", "Pipfile",
            "go.mod", "Cargo.toml", "pom.xml", "build.gradle"
        ]
        for file in dep_files:
            try:
                content = await self.github.get_file_content(owner, repo, file)
                result[file] = content
            except:
                continue
        return result

    def _detect_frameworks(self, dependencies: dict, topics: list[str]) -> list[Framework]:
        """检测框架"""
        frameworks = []

        # 检测package.json中的框架
        if "package.json" in dependencies:
            pkg = json.loads(dependencies["package.json"])
            deps = {**pkg.get("dependencies", {}), **pkg.get("devDependencies", {})}

            framework_map = {
                "react": ("React", "前端框架"),
                "vue": ("Vue.js", "前端框架"),
                "next": ("Next.js", "全栈框架"),
                "nuxt": ("Nuxt.js", "全栈框架"),
                "express": ("Express", "后端框架"),
                "fastapi": ("FastAPI", "后端框架"),
                # ... 更多映射

            for pkg_name, (name, category) in framework_map.items():
                if pkg_name in deps:
                    frameworks.append(Framework(
                        name=name,
                        version=deps[pkg_name],
                        category=category
                    ))
```

### 3.5 RAG代码问答功能

#### 3.5.1 代码索引流程

```python
class RAGService:
    def __init__(self, ai_provider: AIModelProvider):
        self.ai = ai_provider
        self.vector_store = ChromaDBClient()
        self.code_parser = CodeParser()

    async def index_repository(self, owner: str, repo: str, session_id: str):
        """索引仓库代码"""

        # 1. 获取文件树
        file_tree = await self.github.get_tree(owner, repo)

        # 2. 过滤支持的文件类型
        supported_extensions = {'.py', '.js', '.ts', '.java', '.go', '.rs', '.cpp', '.c', '.h'}
        code_files = [
            f for f in file_tree
            if any(f['path'].endswith(ext) for ext in supported_extensions)
            and f['type'] == 'blob'
            and f.get('size', 0) < 100 * 1024  # 100KB限制
        ][:500]  # 最多500个文件

        # 3. 获取文件内容并分块
        chunks = []
        for file in code_files:
            content = await self.github.get_file_content(owner, repo, file['path'])
            file_chunks = self.code_parser.parse(content, file['path'])
            chunks.extend(file_chunks)

        # 4. 生成Embedding并存储
        await self.vector_store.add_documents(
            collection_name=f"{owner}_{repo}_{session_id}",
            documents=chunks
        )

    async def query(self, question: str, session_id: str) -> QAResponse:
        """查询问答"""

        # 1. 检索相关代码片段
        relevant_chunks = await self.vector_store.search(
            collection_name=session_id,
            query=question,
            top_k=5
        )

        # 2. 构建Prompt
        prompt = self._build_qa_prompt(question, relevant_chunks)

        # 3. 生成回答
        answer = await self.ai.generate(prompt)

        # 4. 提取引用
        references = self._extract_references(relevant_chunks)

        return QAResponse(
            answer=answer,
            references=references,
            confidence=0.85  # 可以根据检索分数计算
        )

    def _build_qa_prompt(self, question: str, chunks: list) -> str:
        """构建问答Prompt"""
        context = "\n\n".join([
            f"文件: {chunk['metadata']['file_path']}\n```\n{chunk['content']}\n```"
            for chunk in chunks
        ])

        return f"""基于以下代码上下文，用中文回答用户的问题。

代码上下文：
{context}

用户问题：{question}

请提供详细、准确的回答，并在回答中引用相关的代码文件路径。"""
```

#### 3.5.2 代码解析器

```python
class CodeParser:
    """代码解析器，按函数/类级别分块"""

    def parse(self, content: str, file_path: str) -> list[CodeChunk]:
        """解析代码文件，返回分块结果"""
        ext = Path(file_path).suffix

        if ext == '.py':
            return self._parse_python(content, file_path)
        elif ext in ['.js', '.ts']:
            return self._parse_javascript(content, file_path)
        # ... 其他语言

        # 默认按行分块
        return self._parse_by_lines(content, file_path)

    def _parse_python(self, content: str, file_path: str) -> list[CodeChunk]:
        """解析Python文件"""
        import ast

        chunks = []
        try:
            tree = ast.parse(content)
            for node in ast.walk(tree):
                if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
                    start_line = node.lineno
                    end_line = node.end_lineno
                    chunk_content = '\n'.join(content.split('\n')[start_line-1:end_line])
                    chunks.append(CodeChunk(
                        content=chunk_content,
                        file_path=file_path,
                        line_start=start_line,
                        line_end=end_line,
                        type='function' if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) else 'class'
                    ))
        except SyntaxError:
            # 解析失败时按行分块
            return self._parse_by_lines(content, file_path)

        return chunks if chunks else self._parse_by_lines(content, file_path)
```

## 7. 安全设计

### 7.1 输入验证

```python
from pydantic import BaseModel, validator, HttpUrl

class AnalyzeRequest(BaseModel):
    url: str
    mode: str = "quick"

    @validator('url')
    def validate_github_url(cls, v):
        import re
        pattern = r'^https?://github\.com/[a-zA-Z0-9-]+/[a-zA-Z0-9-._]+$'
        if not re.match(pattern, v):
            raise ValueError('Invalid GitHub repository URL')
        return v

    @validator('mode')
    def validate_mode(cls, v):
        if v not in ['quick', 'deep']:
            raise ValueError('Mode must be "quick" or "deep"')
        return v
```

### 7.3 速率限制

```python
from fastapi import Request, HTTPException
from datetime import datetime
import redis

rate_limit_redis = redis.Redis()

async def rate_limit_middleware(request: Request, call_next):
    client_ip = request.client.host
    key = f"rate_limit:{client_ip}"

    current = rate_limit_redis.get(key)
    if current and int(current) > 100:  # 每分钟100次
        raise HTTPException(status_code=429, detail="Too many requests")

    rate_limit_redis.incr(key)
    rate_limit_redis.expire(key, 60)

    response = await call_next(request)
    return response
```
