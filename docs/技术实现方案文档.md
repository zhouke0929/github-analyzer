```python
# 使用asyncio.gather并发执行多个任务
async def run_analysis(analysis_id, owner, repo, repo_info):
    # 阶段1：并发获取基础数据
    readme_task = github_service.get_readme(owner, repo)
    tech_stack_task = tech_stack_service.analyze(owner, repo)
    
    readme_content = await readme_task
    
    # 阶段2：并发执行AI分析

    # 等待所有任务完成

        translate_task,
        summary_task,
        tech_stack_task,
        architecture_task,
        issues_task
    )
```

#### 3.2.1 模型抽象层设计

```python
class AIService:
    """AI服务封装，支持OpenAI兼容接口"""
    
    def __init__(self):
        self._client = None
        self._model = None
    
    def _get_client(self) -> tuple[AsyncOpenAI, str]:
        """动态获取客户端和模型"""
        settings = get_settings()
        if self._client is None or self._model != settings.openai_model:
            self._client = AsyncOpenAI(
                api_key=settings.openai_api_key,
                base_url=settings.openai_base_url
            )
            self._model = settings.openai_model
        return self._client, self._model
    
    async def generate(self, prompt: str, max_tokens: int = 4096) -> str:
        """生成文本"""
        client, model = self._get_client()
        response = await client.chat.completions.create(
            model=model,
            messages=[{"role": "user", "content": prompt}],
            max_tokens=max_tokens,
            temperature=0.3
        )
        return response.choices[0].message.content
    
    async def generate_json(self, prompt: str, max_tokens: int = 4096) -> dict:
        """生成JSON格式结果"""
        client, model = self._get_client()
        response = await client.chat.completions.create(
            model=model,
            messages=[{"role": "user", "content": prompt}],
            max_tokens=max_tokens,
            temperature=0.3,
            response_format={"type": "json_object"}
        )
        return json.loads(response.choices[0].message.content)
```

```python
from langchain.agents import create_agent, AgentExecutor
from langchain.tools import Tool
from langchain.memory import ConversationBufferMemory
from langchain_core.prompts import PromptTemplate

class CodeAgent:
    """代码问答 Agent（基于 LangChain v1）"""
    
    def __init__(self, llm, vector_store, analysis_data):
        self.llm = llm
        self.vector_store = vector_store
        self.analysis_data = analysis_data
        self.tools = self._init_tools()
        self.memory = ConversationBufferMemory(
            memory_key="chat_history",
            return_messages=True
        )
        self.agent = self._create_agent()
    
    def _init_tools(self):
        """初始化工具"""
        return [
            Tool(
                name="search_code",
                func=self._search_code,
                description="在代码库中精确搜索关键词、函数名、变量名"
            ),
            Tool(
                name="find_definition",
                func=self._find_definition,
                description="查找函数、类、变量的定义位置"
            ),
            Tool(
                name="semantic_search",
                func=self._semantic_search,
                description="语义搜索，理解自然语言问题并检索相关代码"
            ),
            Tool(
                name="get_project_overview",
                func=self._get_project_overview,
                description="获取项目整体概览（技术栈、架构、摘要）"
            )
        ]
    
    def _create_agent(self):
        """创建 LangChain Agent（使用最新 API）"""
        # 注意：LangChain v1 使用 create_agent 替代 create_react_agent
        agent = create_agent(
            model=self.llm,
            tools=self.tools,
            prompt="你是一个代码问答助手，帮助用户理解和分析代码库。"
        )
        
        return AgentExecutor(
            agent=agent,
            tools=self.tools,
            memory=self.memory,
            verbose=True,
            handle_parsing_errors=True
        )
    
    def run(self, question: str) -> str:
        """运行 Agent"""
        result = self.agent.invoke({"input": question})
        return result["output"]
```

#### 3.3.6 LangChain + ChromaDB 集成（最新 API）

```python
from langchain_chroma import Chroma
from langchain_openai import OpenAIEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_core.documents import Document

class RAGService:
    """RAG 服务（基于 LangChain v1 + ChromaDB）"""
    
    def __init__(self, persist_dir: str = "./data/chroma"):
        self.embeddings = OpenAIEmbeddings()
        self.persist_dir = persist_dir
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000,
            chunk_overlap=200,
            separators=["\nclass ", "\ndef ", "\n\n", "\n", " "]
        )
    
    def index_code(self, owner: str, repo: str, files: list):
        """索引代码文件"""
        documents = []
        for file_path, content in files:
            # 创建 Document 对象
            doc = Document(
                page_content=content,
                metadata={
                    "file_path": file_path,
                    "owner": owner,
                    "repo": repo
                }
            )
            # 按函数/类分块
            chunks = self.text_splitter.split_documents([doc])
            documents.extend(chunks)
        
        # 存储到 ChromaDB
        vectorstore = Chroma.from_documents(
            documents=documents,
            embedding=self.embeddings,
            persist_directory=self.persist_dir
        )
        
        return vectorstore
    
    def search(self, query: str, owner: str, repo: str, k: int = 5):
        """语义搜索"""
        vectorstore = Chroma(
            persist_directory=self.persist_dir,
            embedding_function=self.embeddings
        )
        
        results = vectorstore.similarity_search(
            query=query,
            k=k,
            filter={"owner": owner, "repo": repo}
        )
        
        return results
```

#### 3.3.7 Agent 实现示例

```python
class CodeAgent:
    """代码问答 Agent"""
    
    def __init__(self, ai_service, vector_store, github_service):
        self.ai = ai_service
        self.vector_store = vector_store
        self.github = github_service
        self.tools = self._init_tools()
    
    async def run(self, question: str, owner: str, repo: str) -> str:
        """运行 Agent"""
        # 1. 意图识别
        intent = await self._identify_intent(question)
        
        # 2. 选择工具执行
        if intent == "exact_search":
            results = await self._search_code(question, owner, repo)
        elif intent == "semantic_search":
            results = await self._semantic_search(question, owner, repo)
        else:
            results = await self._get_analysis_result(owner, repo)
        
        # 3. 生成回答
        answer = await self._generate_answer(question, results)
        
        return answer
    
    async def _identify_intent(self, question: str) -> str:
        """识别问题意图"""
        prompt = f"""判断以下问题的类型：
        - exact_search: 查找具体的函数、变量、代码
        - semantic_search: 理解功能、架构、设计
        - analysis: 项目概览、技术栈、架构
        
        问题：{question}
        
        只返回类型名称："""
        
        return await self.ai.generate(prompt)
```

## 8. 安全设计

### 8.1 输入验证

```python
from pydantic import BaseModel, field_validator

class AnalyzeRequest(BaseModel):
    url: str
    
    @field_validator('url')
    @classmethod
    def validate_github_url(cls, v: str) -> str:
        import re
        pattern = r'^https?://github\.com/[a-zA-Z0-9._-]+/[a-zA-Z0-9._-]+$'
        if not re.match(pattern, v.rstrip('/')):
            raise ValueError('请输入有效的GitHub仓库URL')
        return v.rstrip('/')
```
